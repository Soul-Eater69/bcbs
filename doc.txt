vc = m[m["in_scope"] & ~m["name"].str.contains(r"I0{6,7}\b|template", case=False, regex=True)]
pats = {
    "value card": r"value\s*card",
    "idea card":  r"idea\s*card",
    "value case": r"value\s*case",   # for reference only, see note below
}
for label, p in pats.items():
    hit = vc["name"].str.contains(p, case=False, regex=True)
    print(f"{label:11} files={hit.sum():4}  ERs={vc.loc[hit, 'er'].nunique()}")

both = vc["name"].str.contains(r"value\s*card|idea\s*card", case=False, regex=True)
print(f"value OR idea card  files={both.sum()}  ERs={vc.loc[both, 'er'].nunique()}")
